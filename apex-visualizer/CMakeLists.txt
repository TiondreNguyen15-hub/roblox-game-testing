if(WIN32)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/imgui.cpp")
  set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
 else()
  include(FetchContent)
  FetchContent_Declare(imgui DOWNLOAD_EXTRACT_TIMESTAMP TRUE URL https://codeload.github.com/ocornut/imgui/tar.gz/refs/tags/v1.91.9b URL_HASH SHA256=8e1bbc76c71d74fef2fb85db7e7ca8eba13d6a86623c54992b60162db554ffdb)
  FetchContent_MakeAvailable(imgui)
  set(IMGUI_DIR "${imgui_SOURCE_DIR}")
 endif()
 add_library(imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp ${IMGUI_DIR}/backends/imgui_impl_win32.cpp)
 target_include_directories(imgui PUBLIC ${IMGUI_DIR} ${IMGUI_DIR}/backends)
 target_compile_definitions(imgui PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
 if(MINGW)
  set(D3D_COMPILER_LIBRARY d3dcompiler_47)
 else()
  set(D3D_COMPILER_LIBRARY d3dcompiler)
 endif()
 target_link_libraries(imgui PUBLIC d3d11 dxgi ${D3D_COMPILER_LIBRARY} dwmapi)
 add_library(apex_hud STATIC src/hud_service.cpp)
 target_include_directories(apex_hud PUBLIC include)
 target_compile_definitions(apex_hud PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(apex_hud PUBLIC apex_core)
 if(MSVC)
  target_sources(apex_hud PRIVATE src/hud_ocr_winrt.cpp)
  target_link_libraries(apex_hud PUBLIC windowsapp)
 else()
  target_sources(apex_hud PRIVATE src/hud_ocr_unavailable.cpp)
 endif()
 add_executable(apex_visualizer WIN32 src/main.cpp src/desktop_overlay.cpp src/enemy_service.cpp src/capture.cpp src/capture_directshow.cpp src/menu.cpp src/renderer.cpp src/visuals.cpp src/system.cpp src/hardware.cpp src/frame_pacer.cpp src/compute.cpp src/runtime_dependencies.cpp)
 target_compile_definitions(apex_visualizer PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_include_directories(apex_visualizer PRIVATE third_party)
 target_link_libraries(apex_visualizer PRIVATE apex_core apex_hud imgui shell32 comdlg32 pdh bcrypt mf mfplat mfreadwrite mfuuid ole32 oleaut32 strmiids)
 add_custom_command(TARGET apex_visualizer POST_BUILD
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/models" "$<TARGET_FILE_DIR:apex_visualizer>/models"
  COMMAND ${CMAKE_COMMAND} -E copy_directory "${CMAKE_CURRENT_SOURCE_DIR}/runtime" "$<TARGET_FILE_DIR:apex_visualizer>/runtime")
 if(MINGW)
  target_link_options(apex_visualizer PRIVATE -municode -Wl,--subsystem,windows)
 endif()
endif()
