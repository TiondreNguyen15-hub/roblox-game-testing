cmake_minimum_required(VERSION 3.24)
project(ApexVisualization VERSION 1.0.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
include(CTest)
add_library(apex_core src/settings.cpp src/config.cpp src/simulation.cpp)
target_include_directories(apex_core PUBLIC include)
if(MSVC)
 target_compile_options(apex_core PRIVATE /W4 /permissive-)
else()
 target_compile_options(apex_core PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(BUILD_TESTING)
 add_executable(apex_tests tests/core_tests.cpp)
 target_link_libraries(apex_tests PRIVATE apex_core)
 add_test(NAME apex_core_modules COMMAND apex_tests)
endif()
if(WIN32)
 if(EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/imgui.cpp")
  set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
 else()
  include(FetchContent)
  FetchContent_Declare(imgui DOWNLOAD_EXTRACT_TIMESTAMP TRUE URL https://codeload.github.com/ocornut/imgui/tar.gz/refs/tags/v1.91.9b URL_HASH SHA256=8e1bbc76c71d74fef2fb85db7e7ca8eba13d6a86623c54992b60162db554ffdb)
  FetchContent_MakeAvailable(imgui)
  set(IMGUI_DIR "${imgui_SOURCE_DIR}")
 endif()
 add_library(imgui STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp ${IMGUI_DIR}/backends/imgui_impl_dx11.cpp ${IMGUI_DIR}/backends/imgui_impl_win32.cpp)
 target_include_directories(imgui PUBLIC ${IMGUI_DIR} ${IMGUI_DIR}/backends)
 target_compile_definitions(imgui PUBLIC UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
 if(MINGW)
  set(D3D_COMPILER_LIBRARY d3dcompiler_47)
 else()
  set(D3D_COMPILER_LIBRARY d3dcompiler)
 endif()
 target_link_libraries(imgui PUBLIC d3d11 dxgi ${D3D_COMPILER_LIBRARY} dwmapi)
 add_executable(apex_visualizer WIN32 src/main.cpp src/menu.cpp src/renderer.cpp src/visuals.cpp src/system.cpp)
 target_compile_definitions(apex_visualizer PRIVATE UNICODE _UNICODE NOMINMAX WIN32_LEAN_AND_MEAN)
 target_link_libraries(apex_visualizer PRIVATE apex_core imgui shell32)
 if(MINGW)
  target_link_options(apex_visualizer PRIVATE -municode -Wl,--subsystem,windows)
 endif()
endif()
# Native/offscreen ImGui rendering tests also run on Linux when dependencies are vendored.
if(BUILD_TESTING AND (TARGET imgui OR EXISTS "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui/imgui.cpp"))
 if(NOT TARGET imgui)
  set(IMGUI_DIR "${CMAKE_CURRENT_SOURCE_DIR}/third_party/imgui")
  add_library(imgui_cpu STATIC ${IMGUI_DIR}/imgui.cpp ${IMGUI_DIR}/imgui_draw.cpp ${IMGUI_DIR}/imgui_tables.cpp ${IMGUI_DIR}/imgui_widgets.cpp)
  target_include_directories(imgui_cpu PUBLIC ${IMGUI_DIR})
  set(TEST_IMGUI imgui_cpu)
 else()
  set(TEST_IMGUI imgui)
 endif()
 add_executable(apex_visuals_tests tests/visuals_tests.cpp src/visuals.cpp)
 target_link_libraries(apex_visuals_tests PRIVATE apex_core ${TEST_IMGUI})
 add_test(NAME apex_imgui_visuals COMMAND apex_visuals_tests)
endif()
if(BUILD_TESTING AND (TARGET imgui OR TARGET imgui_cpu))
 add_executable(apex_menu_tests tests/menu_tests.cpp src/menu.cpp src/system.cpp)
 target_link_libraries(apex_menu_tests PRIVATE apex_core ${TEST_IMGUI})
 add_test(NAME apex_menu_system COMMAND apex_menu_tests)
endif()
option(APEX_BUILD_PREVIEW "Build CPU documentation preview utility" OFF)
if(APEX_BUILD_PREVIEW AND (TARGET imgui OR TARGET imgui_cpu))
 add_executable(apex_preview tools/offscreen_preview.cpp src/menu.cpp src/system.cpp src/visuals.cpp)
 target_link_libraries(apex_preview PRIVATE apex_core ${TEST_IMGUI})
endif()
